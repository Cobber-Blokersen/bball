This was on Fedora 44

# Clean up
podman rm -f bball

# Build
podman build -t bball .

# Create a named volume
podman volume create bball-data

# Run the new container with the same volume
podman run -d --name bball \
  --network slirp4netns:port_handler=slirp4netns \
  -p 8002:8002 \
  -v bball-data:/app/data:Z \
  --restart unless-stopped \
  bball
